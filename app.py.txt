def slow_print(text, delay=0.01):
    """Prints text with a slight delay to simulate an AI typing."""
    for char in text:
        sys.stdout.write(char)
        sys.stdout.flush()
        time.sleep(delay)
    print()

def get_bot_response(user_input):
    """Processes user input and returns the appropriate travel guide response."""
    user_input = user_input.lower().strip()

    # Handle greetings
    if user_input in ["hi", "hello", "hey", "hola"]:
        return "Hello! Where are you planning to travel next? (Try: Paris, Tokyo, New York, Rome)"

    # Handle goodbyes
    if user_input in ["bye", "exit", "quit", "goodbye"]:
        return "Safe travels! Goodbye!"

    # Search for destination keywords in the user's input
    for city, info in TRAVEL_DATA.items():
        if city in user_input:
            if "food" in user_input or "eat" in user_input:
                return f"In {city.title()}, you must try: {info['food']}"
            elif "attraction" in user_input or "see" in user_input or "visit" in user_input:
                return f"Top spots to see in {city.title()}: {info['attractions']}"
            elif "tip" in user_input or "advice" in user_input:
                return f"Insider tip for {city.title()}: {info['tip']}"
            else:
                return (f"Welcome to the {city.title()} guide ({info['country']})!\n"
                        f"📍 Top Attractions: {info['attractions']}\n"
                        f"🍔 Famous Food: {info['food']}\n"
                        f"💡 Quick Tip: {info['tip']}\n\n"
                        f"Ask me specifically about '{city} food' or '{city} tips' for more details!")

     # Help message if keyword isn't found
    return ("I can help you with destinations like Paris, Tokyo, New York, and Rome.\n"
            "Try asking: 'Tell me about Tokyo' or 'What is the food like in Paris?'")

def main():
    print("=============================================")
    print("🌍 WELCOME TO YOUR AI TRAVEL GUIDE CHATBOT 🌍")
    print("=============================================")
    print("Type 'exit' or 'bye' at any time to quit.\n")
    
    slow_print("AI: Hello! I am your virtual travel guide. What city is on your bucket list today?")
    
    while True:
        try:
            user_input = input("\nYou: ")
            if not user_input.strip():
                continue
                
            response = get_bot_response(user_input)
            
            # Print bot response with a stylized layout
            print("AI: ", end="")
            slow_print(response)
            
            if user_input.lower().strip() in ["bye", "exit", "quit"]:
                break
                
        except (KeyboardInterrupt, EOFError):
            print("\nAI: Safe travels! Goodbye!")
            break

main()
